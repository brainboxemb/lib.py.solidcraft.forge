"""Qualification fixture for SolidCraft 0.45 native parametric Slots."""


class SlotFamily:
    width = 160.0
    depth = 120.0
    thickness = 5.0

    slot_width = 8.0
    straight_y = -35.0

    c2c_start_x = -60.0
    c2c_end_x = -40.0

    overall_start_x = -10.0
    overall_end_x = 20.0

    center_point_x = 55.0
    center_point_end_x = 65.0

    def build(self):
        body = Box(
            self.width,
            self.depth,
            self.thickness,
            anchor=Anchor.BOTTOM,
        )

        with Sketch(name="Slot family", on=body) as slots:
            SlotCenterToCenter2D(
                (self.c2c_start_x, self.straight_y),
                (self.c2c_end_x, self.straight_y),
                self.slot_width,
            )
            SlotOverall2D(
                (self.overall_start_x, self.straight_y),
                (self.overall_end_x, self.straight_y),
                self.slot_width,
            )
            SlotCenterPoint2D(
                (self.center_point_x, self.straight_y),
                (self.center_point_end_x, self.straight_y),
                self.slot_width,
            )
            SlotArcThreePoint2D(
                (-50, 25),
                (-35, 38),
                (-20, 25),
                self.slot_width,
            )
            SlotArcCenter2D(
                (40, 25),
                (52, 25),
                self.slot_width,
                sweep=120,
            )

        return body - slots.extrude(Depth.THROUGH_ALL)

Goal

Qualify all five SolidCraft 0.45 Slot construction methods as direct Forge sketch profiles, and prove the native parametric contracts that Forge exposes.

The three straight Slot variants keep axis-aligned length and position editable. All five Slot variants keep width editable. Forge must not approximate any Slot with rounded rectangles or hand-built curves.

Forge model

One 160 × 120 × 5 mm plate contains five Slot profiles in one named sketch.

Baseline geometry intentionally matches the historical qualification:

  • width: 8 mm;
  • straight-Slot Y: -35 mm;
  • centre-to-centre Slot centres: X=-60 .. -40;
  • overall Slot endpoints: X=-10 .. 20;
  • centre-point Slot: centre X=55, end-centre X=65;
  • both arc Slot paths retain fixed authored points.

The lower-case straight-Slot coordinates and slot_width are normal editable Forge model parameters.

How Forge builds it in SolidCraft

Fixed authored Slot points still use the direct native sketch.slot.* creator.

For the three straight Slots, symbolic authored endpoints are evaluated only for numeric seed geometry. Forge then uses native sketch.draw.dimensioned with the SolidCraft 0.45 length role, constrains the returned native centre_line horizontal/vertical and dimensions the returned native middle point from the sketch origin.

The direct Slot point fields therefore remain numeric while native SolidCraft dimensions retain the editable length and position expressions.

The two arc Slots keep fixed authored points but share the same editable native width expression.

SolidCraft result

SolidCraft saves:

  • one plate body;
  • one solved Slot family sketch with five closed native Slot profiles;
  • one error-free through-cut extrusion.

The native timeline must not contain sketch.rectangle.rounded as a Slot substitute.

Saved-design test

Without rerunning Forge Python, change the saved SolidCraft parameters:

  • slot_width: 8 -> 10 mm;
  • straight_y: -35 -> -20 mm;
  • centre-to-centre end X: -40 -> -35 mm;
  • overall end X: 20 -> 25 mm;
  • centre-point end X: 65 -> 70 mm.

The saved design must rebuild solved native Slot geometry with the new width, straight lengths and position. The body envelope stays the same while the through-cut volume changes deterministically.

Geometry evidence

Geometry-rationale provenance: the unchanged baseline envelope/volume contract descends from qualification/geometry_contracts.py at commit be40b09e7454a90e87d83768a9e7453c52ba5d76; the new mutated-volume expectation is derived directly from the five native Slot dimensions declared in this case.

Bounds sufficient: false

The five Slot cuts do not change the plate envelope. Baseline volume and 3D slot-wall probes remain required. The saved-design mutation additionally checks the mutated parameter values, solved sketch state and resulting body volume.

import math
import cadquery as cq

CUT_HEIGHT = 9
CUT_Z = -2
SLOT_WIDTH = 8.0

def arc_band_points(center, radius, start_deg, sweep_deg, width, steps=96):
    cx, cy = center
    half = width / 2
    angles = [
        math.radians(start_deg + sweep_deg * i / steps)
        for i in range(steps + 1)
    ]
    outer = [
        [cx + (radius + half) * math.cos(a), cy + (radius + half) * math.sin(a)]
        for a in angles
    ]
    inner = [
        [cx + (radius - half) * math.cos(a), cy + (radius - half) * math.sin(a)]
        for a in reversed(angles)
    ]
    start = [
        cx + radius * math.cos(angles[0]),
        cy + radius * math.sin(angles[0]),
    ]
    end = [
        cx + radius * math.cos(angles[-1]),
        cy + radius * math.sin(angles[-1]),
    ]
    return outer + inner, start, end


def straight_slot(center_x, center_y, overall_length, width=SLOT_WIDTH):
    return (
        cq.Workplane("XY")
        .center(center_x, center_y)
        .slot2D(overall_length, width)
        .extrude(CUT_HEIGHT)
        .translate((0, 0, CUT_Z))
    )

def arc_slot(center, radius, start_deg, sweep_deg, width=SLOT_WIDTH):
    points, start, end = arc_band_points(
        center, radius, start_deg, sweep_deg, width
    )
    band = (
        cq.Workplane("XY")
        .polyline(points)
        .close()
        .extrude(CUT_HEIGHT)
        .translate((0, 0, CUT_Z))
    )
    start_cap = (
        cq.Workplane("XY")
        .center(start[0], start[1])
        .circle(width / 2)
        .extrude(CUT_HEIGHT)
        .translate((0, 0, CUT_Z))
    )
    end_cap = (
        cq.Workplane("XY")
        .center(end[0], end[1])
        .circle(width / 2)
        .extrude(CUT_HEIGHT)
        .translate((0, 0, CUT_Z))
    )
    return band.union(start_cap).union(end_cap)

plate = cq.Workplane("XY").box(160, 120, 5, centered=(True, True, False))

# overall lengths: 28, 30 and 28 mm respectively.
cutters = straight_slot(-50, -35, 28)
cutters = cutters.union(straight_slot(5, -35, 30))
cutters = cutters.union(straight_slot(55, -35, 28))

three_center = (-35.0, 22.846153846153847)

cutters = cutters.union(

    )
)
cutters = cutters.union(arc_slot((40, 25), 12, 0, 120))

result = plate.cut(cutters)
model = cq.Assembly()
model.add(result, name="slot_family")

def polygon_solid(points):
    min_x = min(p[0] for p in points)
    max_x = max(p[0] for p in points)
    min_y = min(p[1] for p in points)
    max_y = max(p[1] for p in points)
    box_center = ((min_x + max_x) / 2, (min_y + max_y) / 2)
    local = [[p[0] - box_center[0], p[1] - box_center[1]] for p in points]
    return (

def arc_slot(center, radius, start_deg, sweep_deg, width=SLOT_WIDTH):
    points, start, end = arc_band_points(
        center, radius, start_deg, sweep_deg, width
    )
    band = polygon_solid(points)
    start_cap = cyl(diameter=width, height=CUT_HEIGHT).translate(
        [start[0], start[1], 2.5]
    )
    end_cap = cyl(diameter=width, height=CUT_HEIGHT).translate(
        [end[0], end[1], 2.5]
    )
    return band | start_cap | end_cap
