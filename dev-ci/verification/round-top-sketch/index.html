TOP Sketch on circular body

What this example makes

One stepped cylindrical body spans Z=0..26 mm. Disc and Hub remain two solved native Sketch/Extrude feature pairs and the Hub is attached to the actual top face of the Disc.

Forge code

This is the complete Forge source used for this example.

"""Native qualification: TOP-attached Sketch on a circular extrusion."""


class RoundTopSketch:
    def build(self):
        with Sketch(name="Disc", plane="XY") as disc:
            Circle2D(d=100)
        body = disc.extrude(10)

        with Sketch(name="Hub", on=body, parent=Anchor.TOP) as hub:
            Circle2D(d=44)

        return body + hub.extrude(16)


RoundTopSketch()

API used

disc = Sketch(... Circle2D ...).extrude(10); with Sketch(on=disc, parent=Anchor.TOP): Circle2D(d=44); body + hub.extrude(16)

import cadquery as cq

disc = cq.Workplane("XY").circle(50).extrude(10)
hub = cq.Workplane("XY").workplane(offset=10).circle(22).extrude(16)
result = disc.union(hub)

model = cq.Assembly()
model.add(result, name="round_top_sketch")

from build123d import Align, BuildPart, Cylinder, Locations

bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    Cylinder(50, 10, align=bottom)
    with Locations((0, 0, 10)):
        Cylinder(22, 16, align=bottom)
