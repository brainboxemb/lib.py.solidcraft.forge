Minimal L-profile

What this example makes

A clean 40 mm L profile with 12 mm arms and 24 mm symmetric extrusion width.

Forge code

This is the complete Forge source used for this example.

@dataclass
class Bracket:
    """Minimal L-profile qualification fixture.

    Deliberately no holes, fillets or secondary features: this fixture exists
    to verify profile geometry, XZ extrusion and later assembly relationships
    without unrelated feature behavior obscuring the result.
    """

    width: float = 24
    arm_length: float = 40
    arm_height: float = 12

    def build(self):
        with Sketch(name="Profile", plane="XZ") as sketch:
            Polygon2D(
                [
                    (0, 0),
                    (self.arm_length, 0),
                    (self.arm_length, self.arm_height),
                    (self.arm_height, self.arm_height),
                    (self.arm_height, self.arm_length),
                    (0, self.arm_length),
                ],
                plane="XZ",
            )
        return sketch.extrude(self.width, sym=True)

The same model is written independently with the reference libraries below. This makes it easier to compare modelling style and the resulting shape.

PythonSCAD

"""PythonSCAD geometry reference for the minimal symmetric L bracket."""

from pythonscad import *

def box_bounds(lo, hi):
    size = [hi[i] - lo[i] for i in range(3)]
    center = [(hi[i] + lo[i]) / 2 for i in range(3)]
    return cube(size, center=True).translate(center)

horizontal = box_bounds([0, -12, -12], [40, 12, 0])
vertical = box_bounds([0, -12, -40], [12, 12, 0])
model = horizontal | vertical

CadQuery

"""CadQuery geometry reference for the minimal symmetric L bracket."""

import cadquery as cq


horizontal = cq.Solid.makeBox(40, 24, 12, cq.Vector(0, -12, -12))
vertical = cq.Solid.makeBox(12, 24, 40, cq.Vector(0, -12, -40))
bracket = horizontal.fuse(vertical)

model = cq.Assembly()
model.add(bracket, name="bracket")
