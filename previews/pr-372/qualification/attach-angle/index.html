Native attachment clocking angle

What this example makes

The Arm sits on the Base TOP face and its 30x10 footprint is quarter-turned to 10x30 in world XY.

Forge code

This is the complete Forge source used for this example.

"""Qualification: attach(angle=...) is a native joint clocking value."""


class Base:
    WIDTH = 60.0
    DEPTH = 60.0
    HEIGHT = 10.0

    def build(self):
        return Box(self.WIDTH, self.DEPTH, self.HEIGHT, anchor=Anchor.BOTTOM)


class Arm:
    LENGTH = 30.0
    WIDTH = 10.0
    HEIGHT = 6.0

    def build(self):
        return Box(self.LENGTH, self.WIDTH, self.HEIGHT, anchor=Anchor.BOTTOM)


class AttachAngle:
    def build(self):
        base = Base()
        arm = Arm().attach(
            base,
            parent=Anchor.TOP,
            child=Anchor.BOTTOM,
            angle=90,
        )
        return base + arm

"""pyBOSL2 reference: mate the arm to TOP and spin it 90 degrees."""

from pybosl2 import Anchor, cuboid

base = cuboid([60, 60, 10]).up(5)
arm = cuboid([30, 10, 6]).up(3)
model = base.attach(
    Anchor.TOP,
    arm,
    child_anchor=Anchor.BOTTOM,
    spin=90,
).realize()

CadQuery

"""CadQuery assembly reference for a 90-degree clocked top attachment."""

import cadquery as cq


base = cq.Solid.makeBox(60, 60, 10, cq.Vector(-30, -30, 0))
arm = cq.Solid.makeBox(30, 10, 6, cq.Vector(-15, -5, 0))
clocked = cq.Location(cq.Vector(0, 0, 10), cq.Vector(0, 0, 1), 90)

model = cq.Assembly()
model.add(base, name="base")
model.add(arm, name="arm", loc=clocked)

build123d

"""build123d reference: a top-mounted arm clocked 90 degrees in XY."""
from build123d import Align, Box, BuildPart, Locations, Rot

bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    Box(60, 60, 10, align=bottom)
    with Locations((0, 0, 10)):
        with Locations(Rot(0, 0, 90)):
            Box(30, 10, 6, align=bottom)

Reference documentation

  • SolidCraft: The joint angle is the persisted native relationship value that clocks the child about the attachment normal.
  • BOSL2: Conceptually comparable to spin/clocking around an attachment normal, while the child geometry remains reusable and neutral.
