scpf/model.py
"""Four vertical L-profiles centred on the four rounded side corners."""


class FrameBlock:
    _width = 120.0
    _depth = 80.0
    _height = 60.0

    def build(self):
        return Box(
            self._width,
            self._depth,
            self._height,
            anchor=Anchor.BOTTOM,
        ).fillet(8, edges=Edges.VERTICAL)


class LProfile:
    ARM_LENGTH = 28.0
    THICKNESS = 4.0
    PROFILE_HEIGHT = 16.0

    def build(self):
        with Sketch(name="Profile", plane="XY") as sketch:
            Polygon2D(
                [
                    (0, 0),
                    (self.ARM_LENGTH, 0),
                    (self.ARM_LENGTH, -self.THICKNESS),
                    (-self.THICKNESS, -self.THICKNESS),
                    (-self.THICKNESS, self.ARM_LENGTH),
                    (0, self.ARM_LENGTH),
                ],
                plane="XY",
            )
        return sketch.extrude(self.PROFILE_HEIGHT)


class FourCornerL:
    frame_width = 120.0
    frame_depth = 80.0
    frame_height = 60.0

    def build(self):
        frame = FrameBlock(
            _width=self.frame_width,
            _depth=self.frame_depth,
            _height=self.frame_height,
        )
        profile = LProfile()

        half_width = half(self.frame_width)
        half_depth = half(self.frame_depth)
        centre_height = half(
            self.frame_height - LProfile.PROFILE_HEIGHT
        )

        front_left = profile.copy(
            placement=Placement(
                x=-half_width,
                y=-half_depth,
                z=centre_height,
            ),
        )
        front_right = profile.copy(
            placement=Placement(
                x=half_width,
                y=-half_depth,
                z=centre_height,
                rotation=(Z, 90),
            ),
        )
        rear_right = profile.copy(
            placement=Placement(
                x=half_width,
                y=half_depth,
                z=centre_height,
                rotation=(Z, 180),
            ),
        )
        rear_left = profile.copy(
            placement=Placement(
                x=-half_width,
                y=half_depth,
                z=centre_height,
                rotation=(Z, 270),
            ),
        )

        return frame + front_left + front_right + rear_right + rear_left

Engineering explanation

Four centred vertical L-profiles

Documentation provenance: the engineering text below is copied verbatim from the last canonical qualification/cases.py source before its removal, commit be40b09e7454a90e87d83768a9e7453c52ba5d76.

Legacy group: components

Saved-design required: true

Goal

Place one short vertical angle profile around each of the four rounded side corners of a rectangular frame. The assembly calculates each occurrence pose directly from frame dimensions.

Forge API

LProfile.copy(placement=Placement(x=…, y=…, z=…, rotation=(Z, …)))

Forge model

One rounded parametric FrameBlock plus four occurrences of one neutral vertical LProfile definition. Position and clocking belong to the assembly; there is no Attachment.

How Forge builds it in SolidCraft

Forge binds the occurrence ids returned by component.create/occurrence.copy to replay variables. Numeric unrotated copies use occurrence.copy translate directly; symbolic positions and rotations use a later occurrence.move addressed by the bound id.

SolidCraft result

One vertical angle profile wraps each front-left, front-right, rear-right and rear-left rounded side corner. All four occupy Z=22..38 on the 60 mm frame.

Saved-design test

Change frame_width/depth from 120x80 to 160x120 mm in the saved .sc design; all four profiles follow the calculated corner positions while remaining centred at the same frame height.

Geometry evidence

Geometry-rationale provenance: copied verbatim from qualification/geometry_contracts.py at commit be40b09e7454a90e87d83768a9e7453c52ba5d76.

Bounds sufficient: true

This case adds placement of four already-qualified L-profile components around the frame. The component profile itself is qualified by the bracket case; these bounds describe this case’s additional geometry.

Reference notes

  • SolidCraft — component.create/occurrence.copy return real occurrence ids; the replay executor binds those ids before later parametric occurrence.move commands.
  • BOSL2 — The neutral angle profile is reused while the assembly owns position and orientation.

cadquery/model.py
"""CadQuery reference for four short L profiles around a rounded frame."""

import cadquery as cq


frame = (
    cq.Workplane("XY")
    .box(120, 80, 60, centered=(True, True, False))
    .edges("|Z")
    .fillet(8)
    .val()
)

horizontal = cq.Solid.makeBox(32, 4, 16, cq.Vector(-4, -4, -8))
vertical = cq.Solid.makeBox(4, 32, 16, cq.Vector(-4, -4, -8))
profile = horizontal.fuse(vertical)

model = cq.Assembly()
model.add(frame, name="frame")
for index, (x, y, angle) in enumerate([
    (-60, -40, 0),
    (60, -40, 90),
    (60, 40, 180),
    (-60, 40, 270),
]):
    loc = (
        cq.Location((x, y, 30))
        * cq.Location(cq.Vector(0, 0, 0), cq.Vector(0, 0, 1), angle)
    )
    model.add(profile, name=f"profile_{index}", loc=loc)

build123d/model.py
"""build123d reference: four reusable L protectors around a rounded frame."""
from build123d import (
    Align,
    Axis,
    BasePartObject,
    Box,
    BuildPart,
    Locations,
    Mode,
    Rot,
    fillet,
)


class LProtector(BasePartObject):
    def __init__(self, mode=Mode.ADD):
        with BuildPart() as profile:
            with Locations((12, -2, 0)):
                Box(32, 4, 16)
            with Locations((-2, 12, 0)):
                Box(4, 32, 16)
        super().__init__(part=profile.part, mode=mode)


bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    Box(120, 80, 60, align=bottom)
    fillet(model.edges().filter_by(Axis.Z), radius=8)

    for x, y, angle in [
        (-60, -40, 0),
        (60, -40, 90),
        (60, 40, 180),
        (-60, 40, 270),
    ]:
        with Locations((x, y, 30)):
            with Locations(Rot(0, 0, angle)):
                LProtector()

pythonscad/model.py
"""PythonSCAD reference: four short L profiles centred on the frame corners."""

from pythonscad import *

frame = cube([120, 80, 60], center=True).up(30).fillet(8)

def l_profile():
    horizontal = cube([32, 4, 16], center=True).translate([12, -2, 0])
    vertical = cube([4, 32, 16], center=True).translate([-2, 12, 0])
    return horizontal | vertical

profile = l_profile()
model = frame
for x, y, angle in [(-60, -40, 0), (60, -40, 90), (60, 40, 180), (-60, 40, 270)]:
    model = model | profile.rotz(angle).translate([x, y, 30])

"""pyBOSL2 geometry reference: four short L profiles on the side corners."""

from pybosl2 import Anchor, cuboid

frame = cuboid([120, 80, 60], rounding=8, edges=Anchor.Z).up(30)

def l_profile():
    horizontal = cuboid([32, 4, 16]).translate([12, -2, 0])
    vertical = cuboid([4, 32, 16]).translate([-2, 12, 0])
    return horizontal | vertical

profile = l_profile()
model = frame
for x, y, angle in [(-60, -40, 0), (60, -40, 90), (60, 40, 180), (-60, 40, 270)]:
    model = model | profile.rotate(angle, [0, 0, 1]).translate([x, y, 30])
