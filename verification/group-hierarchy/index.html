Named native group hierarchy

What this example makes

Two 10 x 2 x 20 mm panels remain nested under PanelGroup and one 35 x 10 x 40 mm frame remains at root; native BOM inspection must preserve that hierarchy.

Forge code

This is the complete Forge source used for this example.

"""Qualification: group(name) becomes a real nested native component."""


class Panel:
    def build(self):
        return Box(10, 2, 20, anchor=Anchor.BOTTOM)


class Frame:
    def build(self):
        # The frame offset is visual fixture geometry, not the grouping API.
        return Box(35, 10, 40, anchor=Anchor.BOTTOM).move(y=20)


class GroupHierarchy:
    def build(self):
        left = Panel()
        panels = (left + left.copy(placement=Placement(x=11))).group("PanelGroup")
        frame = Frame()
        return panels + frame

panel = cube([10, 2, 20], center=True).up(10)
panels = panel | panel.right(11)
frame = cube([35, 10, 40], center=True).up(20).back(20)
model = panels | frame

"""CadQuery assembly reference for grouped panels plus a separate frame.

This compares nested assembly intent and visible geometry. SolidCraft BOM
persistence remains native Forge qualification evidence.
"""

import cadquery as cq


panel = cq.Solid.makeBox(
    10,
    2,
    20,
    cq.Vector(-5, -1, 0),
)
frame = cq.Solid.makeBox(
    35,
    10,
    40,
    cq.Vector(-17.5, -5, 0),
)

panel_group = cq.Assembly(name="PanelGroup")
panel_group.add(panel, name="panel_1", loc=cq.Location((0, 0, 0)))
panel_group.add(panel, name="panel_2", loc=cq.Location((11, 0, 0)))

model = cq.Assembly()
model.add(panel_group, name="panels")
model.add(frame, name="frame", loc=cq.Location((0, 20, 0)))

"""build123d reference: two panels plus a separate frame.

The builder compares visible geometry; SolidCraft-native qualification proves
browser hierarchy and BOM persistence.
"""
from build123d import Align, Box, BuildPart, Locations

bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    with Locations((0, 0, 0), (11, 0, 0)):
        Box(10, 2, 20, align=bottom)
    with Locations((0, 20, 0)):
        Box(35, 10, 40, align=bottom)
