TOP Sketch after chamfer

What this example makes

SolidCraft 0.42 builds one chamfered plate with one centered through hole. The Top hole Sketch remains a solved feature attached to the actual post-chamfer top face.

Forge code

This is the complete Forge source used for this example.

"""TOP-attached Sketch after a native chamfer.

The chamfer intentionally precedes the attached Sketch. This exercises replay
selection of the resulting planar +Z face through model.faces.
"""


class TopSketchAfterChamfer:
    def build(self):
        plate = Box(100, 70, 10, anchor=Anchor.BOTTOM)
        plate = plate.chamfer(8, edges=Edges.VERTICAL)

        with Sketch(
            name="Top hole",
            on=plate,
            parent=Anchor.TOP,
        ) as hole:
            Circle2D(d=12)

        return plate - hole.extrude(Depth.THROUGH_ALL)


TopSketchAfterChamfer()

API used

Box(...).chamfer(..., edges=Edges.VERTICAL); with Sketch(on=plate, parent=Anchor.TOP): Circle2D(...); cut through all

"""PythonSCAD geometry reference for a chamfered plate with one through hole."""

from pythonscad import *


CUTTER_SIZE = 100.0
CHAMFER_LINE = 50.0 + 35.0 - 8.0
CUTTER_CENTER = (
    CHAMFER_LINE / 2.0
    + CUTTER_SIZE / (2.0 * (2.0 ** 0.5))
)

plate = cube([100, 70, 10], center=True).translate([0, 0, 5])
for sx, sy in ((1, 1), (-1, 1), (-1, -1), (1, -1)):
    cutter = (
        cube([CUTTER_SIZE, CUTTER_SIZE, 20], center=True)
        .rotate([0, 0, 45])
        .translate([sx * CUTTER_CENTER, sy * CUTTER_CENTER, 5])
    )
    plate = plate - cutter

hole = cylinder(d=12, h=20).translate([0, 0, -5])
model = plate - hole

"""CadQuery reference for a chamfered plate with one through hole."""

import cadquery as cq


plate = (
    cq.Workplane("XY")
    .box(100, 70, 10, centered=(True, True, False))
    .edges("|Z")
    .chamfer(8)
    .faces(">Z")
    .workplane()
    .hole(12)
)

model = cq.Assembly()
model.add(plate, name="top_sketch_after_chamfer")

"""build123d reference for a chamfered plate with one through hole."""

from build123d import (
    Align,
    Axis,
    Box,
    BuildPart,
    BuildSketch,
    Circle,
    Mode,
    chamfer,
    extrude,
)


with BuildPart() as model:
    Box(
        100,
        70,
        10,
        align=(Align.CENTER, Align.CENTER, Align.MIN),
    )
    chamfer(model.edges().filter_by(Axis.Z), length=8)
    with BuildSketch(model.faces().sort_by(Axis.Z)[-1]):
        Circle(6)
    extrude(amount=-10, mode=Mode.SUBTRACT)

How Forge builds it in SolidCraft

Forge emits solid.chamfer before model.faces. Replay selects the largest +Z planar face by plane_normal and area, tolerating only tiny numeric-vector noise from the native kernel, then passes its point/index to face-attached sketch.create.
