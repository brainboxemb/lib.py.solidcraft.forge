Explicit component parameters and shared dependencies

What this example makes

The saved SolidCraft design exposes d1_Panel_width, d2_Panel_2_width and shared_width; changing shared_width updates both directly-bound children and the derived half-width child without rerunning Python.

Forge code

This is the complete Forge source used for this example.

"""Parameter scope qualification: explicit component publication and inherited bindings."""


@component(include=("width",))
class Panel:
    width = 40.0
    depth = 20.0
    HEIGHT = 10.0
    _x = 0.0

    def build(self):
        # _x is fixed fixture layout, not a public occurrence-placement
        # dependency. Keeping it in the body preserves the existing visual
        # qualification while this case stays focused on parameter scope.
        return Box(
            self.width,
            self.depth,
            self.HEIGHT,
            anchor=Anchor.BOTTOM,
        ).move(x=self._x)


class ParameterScope:
    shared_width = 80.0

    def build(self):
        local = Panel(name="local", _x=0)
        override = Panel(width=60.0, name="override", _x=100)

        shared_a = Panel(width=self.shared_width, name="shared_a", _x=220)
        shared_b = shared_a.copy(
            name="shared_b",
            placement=Placement(x=120),
        )

        derived = Panel(
            width=half(self.shared_width),
            name="derived",
            _x=460,
        )

        return local + override + shared_a + shared_b + derived


ParameterScope()

API used

@component(include=('width',)) class Panel; Panel(width=self.shared_width); Panel(width=half(self.shared_width))

PythonSCAD

"""PythonSCAD geometry reference for Forge parameter-scope qualification.

PythonSCAD evaluates these Python values directly; this reference intentionally
does not claim SolidCraft-style saved-design parameter bindings.
"""

from pythonscad import *

shared_width = 80.0

def panel(width):
    return cube([width, 20, 10], center=True).up(5)

model = (
    panel(40)
    | panel(60).right(100)
    | panel(shared_width).right(220)
    | panel(shared_width).right(340)
    | panel(shared_width / 2).right(460)
)

CadQuery

"""CadQuery geometry reference for explicit parameter-scope qualification.

CadQuery evaluates these Python values directly. Saved SolidCraft parameter
publication and mutation remain native Forge qualification evidence.
"""

import cadquery as cq


shared_width = 80.0


def panel(width):
    return cq.Solid.makeBox(
        width,
        20,
        10,
        cq.Vector(-width / 2, -10, 0),
    )


model = cq.Assembly()
model.add(panel(40), name="fixed_40", loc=cq.Location((0, 0, 0)))
model.add(panel(60), name="fixed_60", loc=cq.Location((100, 0, 0)))
model.add(panel(shared_width), name="shared_1", loc=cq.Location((220, 0, 0)))
model.add(panel(shared_width), name="shared_2", loc=cq.Location((340, 0, 0)))
model.add(panel(shared_width / 2), name="derived", loc=cq.Location((460, 0, 0)))

build123d

"""build123d geometry reference: fixed/shared/derived panel widths.

Persisted native SolidCraft parameter changes are qualified separately.
"""
from build123d import Align, Box, BuildPart, Locations

shared_width = 80.0
bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    for x, width in [
        (0, 40),
        (100, 60),
        (220, shared_width),
        (340, shared_width),
        (460, shared_width / 2),
    ]:
        with Locations((x, 0, 0)):
            Box(width, 20, 10, align=bottom)

How Forge builds it in SolidCraft

Explicitly included component parameters are namespaced with the internal definition number plus readable component name (for example d1_Panel_width); unincluded child defaults stay fixed; inherited names such as shared_width remain unprefixed and are emitted once even when several child definitions use them.
