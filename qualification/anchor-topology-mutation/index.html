Forge model example

PASS

Explicit topology pick follows parent resize

What this example makes

On the 40x20x20 board the marker is centered on the front-top-left corner selected by the explicit topology pick.

Forge code

This is the complete Forge source used for this example.

"""Qualification: a fixed AnchorTopology pick follows resized parent topology."""


class Board:
    width = 40.0
    height = 20.0
    DEPTH = 20.0

    def anchors(self):
        # Deliberately fixed pick coordinates from the baseline 40x20x20 board.
        # SolidCraft must retain the selected edge/face/corner after later
        # parameter changes; Forge does not recompute these points.
        return {
            "front_top_left_pick": AnchorTopology(
                edge=(-20.0, -10.0, 10.0),
                on=(0.0, -10.0, 10.0),
                corner=(-20.0, -10.0, 20.0),
                z=(0.0, -1.0, 0.0),
                x=(0.0, 0.0, -1.0),
            )
        }

    def build(self):
        return Box(self.width, self.DEPTH, self.height, anchor=Anchor.BOTTOM)


class Marker:
    SIZE = 10.0

    def build(self):
        # Keep the marker on one side of its attachment origin. Native
        # SolidCraft mating then places it fully outside the selected front
        # face, touching the parent exactly at the topology corner.
        return Box(self.SIZE, self.SIZE, self.SIZE).move(z=-half(self.SIZE))


class AnchorTopologyMutation:
    base_width = 40.0
    base_height = 20.0

    def build(self):
        board = Board(
            width=self.base_width,
            height=self.base_height,
        )
        marker = Marker().attach(
            board,
            parent="front_top_left_pick",
            child=AnchorFrame(),
        )
        return board + marker


AnchorTopologyMutation()

API used

anchors() -> AnchorTopology(edge=..., on=..., corner=..., z=..., x=...); attach(parent='front_top_left_pick', child=AnchorFrame())

"""PythonSCAD geometry reference for a marker at the board's front/top/left corner.

The Forge case is stronger: a fixed topology pick must follow a saved-design
resize in SolidCraft without Python recomputing these coordinates.
"""

from pythonscad import *

board = cube([40, 20, 20], center=True).up(10)
marker = cube([10, 10, 10], center=True).translate([-20, -15, 20])
model = board | marker

CadQuery

"""CadQuery geometry reference for the selected board corner and marker.

Forge's stronger contract is persisted native topology identity after a saved
SolidCraft resize; this reference compares the baseline geometry and datum.
"""

import cadquery as cq


board = cq.Solid.makeBox(40, 20, 20, cq.Vector(-20, -10, 0))
marker = cq.Solid.makeBox(10, 10, 10, cq.Vector(-5, -5, -5))

model = cq.Assembly()
model.add(board, name="board")
model.add(marker, name="marker", loc=cq.Location((-20, -15, 20)))

bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    Box(40, 20, 20, align=bottom)
    with Locations((-20, -15, 20)):
        Box(10, 10, 10)

How Forge builds it in SolidCraft

AnchorTopology is inlined into joint.create as edge/on/corner plus optional z/x. No persistent joint.origin and no Forge-side recomputation of the pick coordinates occurs after the saved design is mutated.
