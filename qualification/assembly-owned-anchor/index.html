Forge model example

PASS

Assembly-owned named anchor

What this example makes

The marker sits on the base front-top-right corner while an unrelated companion occurrence remains part of the same assembly.

Forge code

This is the complete Forge source used for this example.

"""Qualification: an assembly-owned named anchor remains associative."""


class Base:
    width = 40.0
    height = 20.0
    DEPTH = 30.0

    def build(self):
        return Box(
            self.width,
            self.DEPTH,
            self.height,
            anchor=Anchor.BOTTOM,
        )


class Companion:
    SIZE = 10.0

    def build(self):
        # Position is not part of this qualification. Keep its old visible
        # location inside this unique component definition.
        return Box(self.SIZE, self.SIZE, self.SIZE).move(x=70, z=self.SIZE / 2)


class Marker:
    def build(self):
        return Box(10, 10, 10)


class AssemblyOwnedAnchor:
    base_width = 40.0
    base_height = 20.0

    def build(self):
        base = Base(
            width=self.base_width,
            height=self.base_height,
            name="base",
        )
        companion = Companion(name="companion")

        frame = (base + companion).with_anchor(
            "inspection_corner",
            base,
            Anchor.FRONT_TOP_RIGHT,
        )
        datum = frame.anchor("inspection_corner")

        marker = Marker().attach(
            datum.target,
            parent=datum.reference,
            child=Anchor.BACK_TOP_RIGHT,
            offset=datum.offset,
        )
        return frame + marker

PythonSCAD

"""PythonSCAD geometry reference for an assembly-owned inspection datum.

Forge additionally qualifies that the named datum belongs to the assembly and
remains associated with the base corner after native parameter mutation.
"""

from pythonscad import *

base = cube([40, 30, 20], center=True).up(10)
companion = cube([10, 10, 10], center=True).translate([70, 0, 5])
marker = cube([10, 10, 10], center=True).translate([15, -20, 15])
model = base | companion | marker

"""CadQuery assembly reference for an assembly-owned inspection datum layout."""

import cadquery as cq


base = cq.Solid.makeBox(40, 30, 20, cq.Vector(-20, -15, 0))
small = cq.Solid.makeBox(10, 10, 10, cq.Vector(-5, -5, -5))

model = cq.Assembly()
model.add(base, name="base")
model.add(small, name="companion", loc=cq.Location((70, 0, 5)))
model.add(small, name="marker", loc=cq.Location((15, -20, 15)))

How Forge builds it in SolidCraft

Reference documentation

  • BOSL2: Semantic attachment names belong naturally to the assembly that owns their meaning rather than to a generic reusable component.
  • SolidCraft: The semantic name disappears during lowering; the persisted native relationship remains topology-backed and associative.
